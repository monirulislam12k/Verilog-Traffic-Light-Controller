
module traffic_light (
    input clk,
    input rst,
    output reg red,
    output reg green,
    output reg yellow
);

    // State definitions
    localparam RED    = 2'b00;
    localparam GREEN  = 2'b01;
    localparam YELLOW = 2'b10;

    reg [1:0] state;
    reg [1:0] next_state;

    // 1. State register
    always @(posedge clk or posedge rst) begin
        if (rst)
            state <= RED;
        else
            state <= next_state;
    end

    // 2. Next-state logic
    always @(*) begin
        next_state = state;

        case (state)
            RED:    next_state = GREEN;
            GREEN:  next_state = YELLOW;
            YELLOW: next_state = RED;

            default: next_state = RED;
        endcase
    end

    // 3. Output logic
    always @(*) begin
        red    = 1'b0;
        green  = 1'b0;
        yellow = 1'b0;

        case (state)
            RED:    red    = 1'b1;
            GREEN:  green  = 1'b1;
            YELLOW: yellow = 1'b1;

            default: red = 1'b1;
        endcase
    end

endmodule
